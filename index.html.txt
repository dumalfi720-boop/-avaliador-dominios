        // Calcular avaliação
        function calcularAvaliacao() {
            const nomeDominio = document.getElementById('nomeDominio').value.trim();
            const extensao = document.getElementById('extensao').value;

            if (!nomeDominio) {
                alert('Por favor, insira o nome do domínio!');
                return;
            }

            let pontuacaoTotal = 0;
            let pesoTotal = 0;
            const avaliacoes = {};

            criterios.forEach(criterio => {
                const valor = parseFloat(document.getElementById(criterio.id).value);
                avaliacoes[criterio.id] = valor;
                pontuacaoTotal += valor * criterio.peso;
                pesoTotal += criterio.peso * 10; // máximo possível
            });

            const pontuacaoFinal = (pontuacaoTotal / pesoTotal * 100).toFixed(1);
            const valorEstimado = calcularValor(nomeDominio, pontuacaoFinal, extensao);

            const avaliacao = {
                id: Date.now(),
                dominio: nomeDominio + extensao,
                pontuacao: pontuacaoFinal,
                valor: valorEstimado,
                data: new Date().toISOString(),
                avaliacoes: avaliacoes
            };

            historico.unshift(avaliacao);
            localStorage.setItem('historicoAvaliacoes', JSON.stringify(historico));

            exibirResultado(avaliacao);
            atualizarDashboard();
        }

        // Calcular valor estimado
        function calcularValor(nome, pontuacao, extensao) {
            let valorBase = 100;

            // Tamanho do domínio
            if (nome.length <= 4) valorBase += 5000;
            else if (nome.length <= 6) valorBase += 2000;
            else if (nome.length <= 8) valorBase += 1000;
            else if (nome.length <= 10) valorBase += 500;

            // Extensão
            const multiplicadoresExtensao = {
                '.com': 3.0,
                '.com.br': 2.0,
                '.io': 2.5,
                '.net': 1.8,
                '.org': 1.5,
                '.app': 2.2,
                '.tech': 1.7,
                '.store': 1.6
            };

            valorBase *= (multiplicadoresExtensao[extensao] || 1.0);

            // Pontuação
            valorBase *= (pontuacao / 50);

            // Palavras-chave valiosas
            const palavrasValiosas = ['app', 'tech', 'digital', 'smart', 'cloud', 'ai', 'data', 'web', 'net', 'pro'];
            palavrasValiosas.forEach(palavra => {
                if (nome.toLowerCase().includes(palavra)) {
                    valorBase *= 1.5;
                }
            });

            return Math.round(valorBase);
        }

        // Gerar insights
        function gerarInsights(avaliacao) {
            const insights = [];
            const nome = avaliacao.dominio.split('.')[0];

            // Análise de tamanho
            if (nome.length <= 5) {
                insights.push({
                    titulo: '✅ Tamanho Excelente',
                    descricao: 'Domínio curto, fácil de digitar e memorizar.'
                });
            } else if (nome.length > 12) {
                insights.push({
                    titulo: '⚠️ Domínio Longo',
                    descricao: 'Considere alternativas mais curtas para melhor memorização.'
                });
            }

            // Análise de pontuação
            if (avaliacao.pontuacao >= 90) {
                insights.push({
                    titulo: '🎯 Excelente Investimento',
                    descricao: 'Domínio premium com alto potencial comercial.'
                });
            } else if (avaliacao.pontuacao >= 70) {
                insights.push({
                    titulo: '👍 Bom Potencial',
                    descricao: 'Domínio sólido com boas características.'
                });
            }

            // Análise de valor
            if (avaliacao.valor >= 10000) {
                insights.push({
                    titulo: '💰 Alto Valor Comercial',
                    descricao: 'Domínio premium com excelente valor de mercado.'
                });
            }

            // Sugestões gerais
            insights.push({
                titulo: '📈 Recomendação',
                descricao: avaliacao.pontuacao >= 80 
                    ? 'Este domínio apresenta características excelentes para branding e SEO.'
                    : 'Considere melhorar aspectos como memorabilidade e relevância.'
            });

            return insights;
        }

        // Classificar pontuação
        function classificarPontuacao(pontuacao) {
            if (pontuacao >= 90) return { texto: 'Excelente', classe: 'badge-success' };
            if (pontuacao >= 80) return { texto: 'Muito Bom', classe: 'badge-success' };
            if (pontuacao >= 70) return { texto: 'Bom', classe: 'badge-warning' };
            if (pontuacao >= 60) return { texto: 'Regular', classe: 'badge-warning' };
            return { texto: 'Precisa Melhorar', classe: 'badge-danger' };
        }

        // Atualizar dashboard
        function atualizarDashboard() {
            if (historico.length === 0) {
                document.getElementById('totalAvaliacoes').textContent = '0';
                document.getElementById('pontuacaoMedia').textContent = '0';
                document.getElementById('valorMedio').textContent = 'R$ 0';
                document.getElementById('melhorAvaliacao').textContent = '0';
                document.getElementById('ultimasAvaliacoes').innerHTML = '<p style="text-align: center; color: var(--gray-600); padding: 40px;">Nenhuma avaliação realizada ainda.</p>';
                return;
            }

            const total = historico.length;
            const pontuacaoMedia = (historico.reduce((sum, item) => sum + parseFloat(item.pontuacao), 0) / total).toFixed(1);
            const valorMedio = Math.round(historico.reduce((sum, item) => sum + item.valor, 0) / total);
            const melhorPontuacao = Math.max(...historico.map(item => parseFloat(item.pontuacao)));

            document.getElementById('totalAvaliacoes').textContent = total;
            document.getElementById('pontuacaoMedia').textContent = pontuacaoMedia;
            document.getElementById('valorMedio').textContent = `R$ ${valorMedio.toLocaleString('pt-BR')}`;
            document.getElementById('melhorAvaliacao').textContent = melhorPontuacao;

            // Últimas 5 avaliações
            const ultimasHtml = historico.slice(0, 5).map(item => {
                const classificacao = classificarPontuacao(item.pontuacao);
                return `
                    <div class="history-item">
                        <div>
                            <div class="history-domain">${item.dominio}</div>
                            <small style="color: var(--gray-600);">
                                ${new Date(item.data).toLocaleDateString('pt-BR')}
                            </small>
                        </div>
                        <div style="text-align: right;">
                            <div class="history-score">${item.pontuacao}</div>
                            <div class="badge ${classificacao.classe}" style="margin-top: 5px;">
                                ${classificacao.texto}
                            </div>
                        </div>

            }).join('');

            criarGraficoDashboard();

        // Filtrar histórico
        function filtrarHistorico() {
            const filtroNome = document.getElementById('filtroNome').value.toLowerCase();
            const filtroExt = document.getElementById('filtroExtensao').value;
            const filtroPont = document.getElementById('filtroPontuacao').value;

            let filtrado = historico.filter(item => {
                let match = true;

                if (filtroNome && !item.dominio.toLowerCase().includes(filtroNome)) {
                    match = false;
                }

                if (filtroExt && !item.dominio.includes(filtroExt)) {
                    match = false;
                }

                if (filtroPont) {
                    const pont = parseFloat(item.pontuacao);
                    switch(filtroPont) {
                        case 'excelente': match = match && pont >= 90; break;
                        case 'muito-bom': match = match && pont >= 80 && pont < 90; break;
                        case 'bom': match = match && pont >= 70 && pont < 80; break;
                        case 'regular': match = match && pont >= 60 && pont < 70; break;
                        case 'baixo': match = match && pont < 60; break;
                    }
                }

                return match;
            });

            const container = document.getElementById('listaHistorico');
            
            if (filtrado.length === 0) {
                container.innerHTML = '<p style="text-align: center; padding: 40px; color: var(--gray-600);">Nenhum resultado encontrado.</p>';
                return;
            }

            const html = filtrado.map(item => {
                const classificacao = classificarPontuacao(item.pontuacao);
                return `
                    <div class="history-item">
                        <div>
                            <div class="history-domain">${item.dominio}</div>
                            <small style="color: var(--gray-600);">
                                ${new Date(item.data).toLocaleString('pt-BR')} • 
                                R$ ${item.valor.toLocaleString('pt-BR')}
                            </small>
                        </div>
                        <div style="display: flex; gap: 10px; align-items: center;">
                            <div class="badge ${classificacao.classe}">${classificacao.texto}</div>
                            <div class="history-score">${item.pontuacao}</div>
                            <button class="btn btn-danger" onclick="excluirAvaliacao(${item.id})" 
                                    style="padding: 8px 12px; font-size: 0.9em;">
                                🗑️
                            </button>
                        </div>
                    </div>
                `;
            }).join('');

            container.innerHTML = html;
        }

        // Excluir avaliação
        function excluirAvaliacao(id) {
            if (!confirm('Tem certeza que deseja excluir esta avaliação?')) return;

            historico = historico.filter(item => item.id !== id);
            localStorage.setItem('historicoAvaliacoes', JSON.stringify(historico));
            carregarHistorico();
            atualizarDashboard();
        }

        // Limpar histórico
        function limparHistorico() {
            if (!confirm('Tem certeza que deseja limpar TODO o histórico? Esta ação não pode ser desfeita!')) return;

            historico = [];
            localStorage.setItem('historicoAvaliacoes', JSON.stringify(historico));
            carregarHistorico();
            atualizarDashboard();
        }

        // Exportar CSV
        function exportarCSV() {
            if (historico.length === 0) {
                alert('Não há dados para exportar!');
                return;
            }

            let csv = 'Domínio,Pontuação,Valor (R$),Data\n';
            
            historico.forEach(item => {
                csv += `${item.dominio},${item.pontuacao},${item.valor},${new Date(item.data).toLocaleString('pt-BR')}\n`;
            });

            const blob = new Blob([csv], { type: 'text/csv' });
            const url = window.URL.createObjectURL(blob);
            const a = document.createElement('a');
            a.href = url;
            a.download = `avaliacoes_dominios_${new Date().getTime()}.csv`;
            a.click();
        }
